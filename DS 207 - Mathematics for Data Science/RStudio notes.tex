x <- c(2,3,3,5,7)
mean(x)

plot(
x, 
rep(1, length(x)), 
pch = 19, 
col="blue",
xlab = "Values", 
ylab = " ", 
yaxt = "n", 
main = "Plot of x with Mean")

points(x, 1, pch = 19, col = "red")
